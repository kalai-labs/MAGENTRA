--------------------------------- rle.py -------------------------------------
"""Run-length codec for short strings.

encode() collapses a run of the same character into <count><char>, and leaves a
run of one as the bare character, because that is shorter.
decode() reverses it.
"""

DIGITS = "0123456789"


def encode(text):
    out = []
    i = 0
    while i < len(text):
        ch = text[i]
        run = 1
        while i + run < len(text) and text[i + run] == ch:
            run += 1
        out.append(ch if run == 1 else str(run) + ch)
        i += run
    return "".join(out)


def decode(text):
    out = []
    i = 0
    while i < len(text):
        digits = ""
        while i < len(text) and text[i] in DIGITS:
            digits += text[i]
            i += 1
        ch = text[i]
        i += 1
        out.append(ch if digits == "" else ch * int(digits))
    return "".join(out)
