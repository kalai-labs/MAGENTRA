def pad(value, width):
    s = str(value)
    return s if len(s) >= width else " " * (width - len(s)) + s

import re

from .util import pad

DAY_RE = re.compile(r"^\d{4}-\d{2}-\d{2}$")


def csv_report(rows):
    clean = []
    for r in rows:
        name = r.get("name")
        cents = r.get("cents")
        day = r.get("day")
        if not isinstance(name, str) or name.strip() == "":
            continue
        if not isinstance(cents, int) or isinstance(cents, bool):
            continue
        if not isinstance(day, str) or not DAY_RE.match(day):
            continue
        clean.append({"name": name.strip(), "cents": cents, "day": day})
    clean.sort(key=lambda r: (r["day"], r["name"]))
    lines = ["name,day,amount"]
    total = 0
    for r in clean:
        total += r["cents"]
        sign = "-" if r["cents"] < 0 else ""
        magnitude = abs(r["cents"])
        money = sign + str(magnitude // 100) + "." + pad(magnitude % 100, 2).replace(" ", "0")
        y, m, d = r["day"].split("-")
        day = d + "/" + m + "/" + y
        name = '"' + r["name"].replace('"', '""') + '"' if ('"' in r["name"] or "," in r["name"]) else r["name"]
        lines.append(name + "," + day + "," + money)
    tsign = "-" if total < 0 else ""
    tmag = abs(total)
    lines.append("TOTAL,," + tsign + str(tmag // 100) + "." + pad(tmag % 100, 2).replace(" ", "0"))
    return "\n".join(lines)

import sys

from report_lib.csv_report import csv_report
from report_lib.tsv_report import tsv_report
from report_lib.pipe_report import pipe_report

ROWS = [
    {"name": "Ada Lovelace", "cents": 125000, "day": "2026-03-14"},
    {"name": " Grace Hopper ", "cents": -4250, "day": "2026-01-02"},
    {"name": 'Kurt "Bit" Godel', "cents": 7, "day": "2026-01-02"},
    {"name": "Alan\tTuring", "cents": 999, "day": "2026-03-14"},
    {"name": "Rear|Admiral", "cents": 100000, "day": "2025-12-31"},
    {"name": "", "cents": 500, "day": "2026-01-01"},
    {"name": "Bad Day", "cents": 500, "day": "2026-1-1"},
    {"name": "Bad Money", "cents": 5.5, "day": "2026-01-01"},
    {"name": "Zoe Zulu", "cents": 0, "day": "2026-03-14"},
    {"name": "Truthy Flag", "cents": True, "day": "2026-01-05"},
]


def main(argv):
    fmt = argv[1] if len(argv) > 1 else None
    if fmt == "csv":
        sys.stdout.write(csv_report(ROWS) + "\n")
    elif fmt == "tsv":
        sys.stdout.write(tsv_report(ROWS) + "\n")
    elif fmt == "pipe":
        sys.stdout.write(pipe_report(ROWS) + "\n")
    else:
        sys.stderr.write("usage: python main.py csv|tsv|pipe\n")
        return 2
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv))
